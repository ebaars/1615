(() => {
  // Text that does not fit its box (inputs and value labels), and elements that actually show scrollbars.
  const canvas = document.createElement('canvas'), ctx = canvas.getContext('2d');
  const textOf = (el) => el.tagName === 'INPUT' ? el.value : (el.innerText || '').trim();
  const clipped = [], scrolls = [];
  for (const el of document.querySelectorAll('input, [class*="psc-cal1615/value"], [class*="psc-cal1615/status"], [class*="psc-cal1615/entry"]')) {
    const cs = getComputedStyle(el);
    const t = textOf(el);
    if (!t || el.offsetParent === null) continue;
    ctx.font = cs.fontWeight + ' ' + cs.fontSize + ' ' + cs.fontFamily;
    const need = ctx.measureText(t).width;
    const have = el.clientWidth - parseFloat(cs.paddingLeft) - parseFloat(cs.paddingRight);
    const needH = parseFloat(cs.lineHeight) || parseFloat(cs.fontSize) * 1.2;
    const haveH = el.clientHeight - parseFloat(cs.paddingTop) - parseFloat(cs.paddingBottom);
    if (need > have + 0.5 || needH > haveH + 0.5) clipped.push(`${el.tagName} "${t}" text ${need.toFixed(0)}x${needH.toFixed(0)} box ${have.toFixed(0)}x${haveH.toFixed(0)}`);
  }
  for (const el of document.querySelectorAll('*')) {
    const cs = getComputedStyle(el);
    const sy = /auto|scroll/.test(cs.overflowY) && el.scrollHeight > el.clientHeight + 1;
    const sx = /auto|scroll/.test(cs.overflowX) && el.scrollWidth > el.clientWidth + 1;
    if ((sx || sy) && el.closest('[data-component="ia.chart.powerchart"]') === null && el.clientHeight < 900)
      scrolls.push(`${sx ? 'H' : ''}${sy ? 'V' : ''} ${el.tagName}.${String(el.className).slice(0, 50)} box ${el.clientWidth}x${el.clientHeight} content ${el.scrollWidth}x${el.scrollHeight} "${(el.innerText || '').trim().replace(/\s+/g, ' ').slice(0, 40)}"`);
  }
  const vw = window.innerWidth + 'x' + window.innerHeight;
  const count = (a) => { const m = {}; a.forEach((x) => m[x] = (m[x] || 0) + 1); return Object.entries(m).map(([k, n]) => `  ${n} x ${k}`).slice(0, 12).join('\n'); };
  return `viewport ${vw}  clipped text: ${clipped.length}  scrollbars: ${scrolls.length}\n${count(clipped)}\n${count(scrolls)}`;
})()
